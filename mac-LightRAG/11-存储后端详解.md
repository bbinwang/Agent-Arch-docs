### 12.2.2 实现原理

```python
class JsonKVStorage(BaseKVStorage):
    def __init__(self, namespace, global_config, **kwargs):
        super().__init__(namespace, global_config)
        self.file_path = os.path.join(
            global_config["working_dir"], f"{namespace}.json"
        )
        self.data = {}
        self._load()

    def _load(self):
        """从 JSON 文件加载数据"""
        if os.path.exists(self.file_path):
            with open(self.file_path, "r") as f:
                self.data = json.load(f)

    def _save(self):
        """保存数据到 JSON 文件"""
        with open(self.file_path, "w") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)

    async def get_by_id(self, id: str) -> dict:
        return self.data.get(id)

    async def upsert(self, data: dict):
        self.data.update(data)
        self._save()
```

### 12.2.3 优缺点

| 优点 | 缺点 |
|------|------|
| 零配置 | 性能差（全量读写）|
| 易于调试（文件可读）| 不支持并发写入 |
| 无外部依赖 | 数据量大时内存占用高 |
| 适合小规模数据 | 无索引，查询慢 |

```python
class NetworkXStorage(BaseGraphStorage):
    def __init__(self, namespace, global_config, **kwargs):
        super().__init__(namespace, global_config)
        self.graph = nx.Graph()
        self.file_path = os.path.join(
            global_config["working_dir"], f"{namespace}.graph.json"
        )
        self._load_graph()

    def _load_graph(self):
        """从 JSON 文件加载图"""
        if os.path.exists(self.file_path):
            with open(self.file_path, "r") as f:
                data = json.load(f)
                self.graph = nx.node_link_graph(data)

    def _save_graph(self):
        """保存图到 JSON 文件"""
        data = nx.node_link_graph(self.graph)
        with open(self.file_path, "w") as f:
            json.dump(data, f)

    async def upsert_node(self, node_id, node_data):
        self.graph.add_node(node_id, **node_data)
        self._save_graph()

    async def upsert_edge(self, source_id, target_id, edge_data):
        self.graph.add_edge(source_id, target_id, **edge_data)
        self._save_graph()

    async def get_node(self, node_id):
        if node_id in self.graph:
            return self.graph.nodes[node_id]
        return None

    async def get_edge(self, source_id, target_id):
        if self.graph.has_edge(source_id, target_id):
            return self.graph.edges[source_id, target_id]
        return None

    async def get_node_edges(self, node_id):
        if node_id in self.graph:
            return list(self.graph.edges(node_id))
        return []
```

| 易于调试 | 不支持分布式 |
| 支持图遍历 | 持久化效率低 |

```python
class NanoVectorDBStorage(BaseVectorStorage):
    def __init__(self, namespace, global_config, embedding_func, **kwargs):
        super().__init__(namespace, global_config, embedding_func)
        self.vectordb = NanoVectorDB(
            embedding_func.embedding_dim,
            storage_file=os.path.join(
                global_config["working_dir"], f"{namespace}.vdb.json"
            )
        )

    async def query(self, query, top_k=10, ids=None):
        if isinstance(query, str):
            query = await self.embedding_func([query])
            query = query[0]

        results = self.vectordb.query(query, top_k=top_k)
        return results

    async def upsert(self, data):
        ids = list(data.keys())
        vectors = [d["vector"] for d in data.values()]
        metadatas = [d.get("metadata", {}) for d in data.values()]

        self.vectordb.upsert(ids, vectors, metadatas)
        self.vectordb.save()
```

#### PGKVStorage

```python
class PGKVStorage(BaseKVStorage):
    def __init__(self, namespace, global_config, **kwargs):
        super().__init__(namespace, global_config)
        self.pool = None
        self.table_name = f"lightrag_kv_{namespace}"

    async def initialize(self):
        self.pool = await asyncpg.create_pool(
            host=os.getenv("POSTGRES_HOST", "localhost"),
            port=int(os.getenv("POSTGRES_PORT", 5432)),
            user=os.getenv("POSTGRES_USER"),
            password=os.getenv("POSTGRES_PASSWORD"),
            database=os.getenv("POSTGRES_DATABASE"),
        )
        # 创建表
        await self.pool.execute(f"""
            CREATE TABLE IF NOT EXISTS {self.table_name} (
                key TEXT PRIMARY KEY,
                value JSONB NOT NULL,
                created_at TIMESTAMP DEFAULT NOW(),
                updated_at TIMESTAMP DEFAULT NOW()
            )
        """)

    async def get_by_id(self, id):
        row = await self.pool.fetchrow(
            f"SELECT value FROM {self.table_name} WHERE key = $1", id
        )
        return row["value"] if row else None

    async def upsert(self, data):
        for key, value in data.items():
            await self.pool.execute(f"""
                INSERT INTO {self.table_name} (key, value)
                VALUES ($1, $2)
                ON CONFLICT (key) DO UPDATE SET
                    value = $2,
                    updated_at = NOW()
            """, key, json.dumps(value))
```

#### PGVectorStorage

```python
class PGVectorStorage(BaseVectorStorage):
    def __init__(self, namespace, global_config, embedding_func, **kwargs):
        super().__init__(namespace, global_config, embedding_func)
        self.table_name = f"lightrag_vector_{namespace}"

    async def initialize(self):
        # 启用 pgvector 扩展
        await self.pool.execute("CREATE EXTENSION IF NOT EXISTS vector")
        # 创建表
        await self.pool.execute(f"""
            CREATE TABLE IF NOT EXISTS {self.table_name} (
                id TEXT PRIMARY KEY,
                vector vector({self.embedding_func.embedding_dim}),
                metadata JSONB,
                created_at TIMESTAMP DEFAULT NOW()
            )
        """)
        # 创建 ANN 索引
        await self.pool.execute(f"""
            CREATE INDEX IF NOT EXISTS idx_{self.table_name}_vector
            ON {self.table_name}
            USING ivfflat (vector vector_cosine_ops)
            WITH (lists = 100)
        """)

    async def query(self, query, top_k=10, ids=None):
        if isinstance(query, str):
            query = await self.embedding_func([query])
            query = query[0]

        vector_str = f"[{','.join(str(x) for x in query)}]"

        rows = await self.pool.fetch(f"""
            SELECT id, metadata, 1 - (vector <=> $1::vector) AS similarity
            FROM {self.table_name}
            ORDER BY vector <=> $1::vector
            LIMIT $2
        """, vector_str, top_k)

        return [
            {"id": row["id"], "metadata": row["metadata"], "distance": row["similarity"]}
            for row in rows
        ]
```

#### PGGraphStorage

```python
class PGGraphStorage(BaseGraphStorage):
    def __init__(self, namespace, global_config, **kwargs):
        super().__init__(namespace, global_config)
        self.nodes_table = f"lightrag_graph_nodes_{namespace}"
        self.edges_table = f"lightrag_graph_edges_{namespace}"

    async def initialize(self):
        # 创建节点表
        await self.pool.execute(f"""
            CREATE TABLE IF NOT EXISTS {self.nodes_table} (
                id TEXT PRIMARY KEY,
                properties JSONB NOT NULL,
                created_at TIMESTAMP DEFAULT NOW(),
                updated_at TIMESTAMP DEFAULT NOW()
            )
        """)
        # 创建边表
        await self.pool.execute(f"""
            CREATE TABLE IF NOT EXISTS {self.edges_table} (
                id TEXT PRIMARY KEY,
                source_id TEXT NOT NULL,
                target_id TEXT NOT NULL,
                properties JSONB NOT NULL,
                created_at TIMESTAMP DEFAULT NOW(),
                FOREIGN KEY (source_id) REFERENCES {self.nodes_table}(id),
                FOREIGN KEY (target_id) REFERENCES {self.nodes_table}(id)
            )
        """)
        # 创建索引
        await self.pool.execute(f"""
            CREATE INDEX IF NOT EXISTS idx_{self.edges_table}_source
            ON {self.edges_table}(source_id)
        """)
        await self.pool.execute(f"""
            CREATE INDEX IF NOT EXISTS idx_{self.edges_table}_target
            ON {self.edges_table}(target_id)
        """)

    async def upsert_node(self, node_id, node_data):
        await self.pool.execute(f"""
            INSERT INTO {self.nodes_table} (id, properties)
            VALUES ($1, $2)
            ON CONFLICT (id) DO UPDATE SET
                properties = $2,
                updated_at = NOW()
        """, node_id, json.dumps(node_data))

    async def upsert_edge(self, source_id, target_id, edge_data):
        edge_id = f"{source_id}||{target_id}"
        await self.pool.execute(f"""
            INSERT INTO {self.edges_table} (id, source_id, target_id, properties)
            VALUES ($1, $2, $3, $4)
            ON CONFLICT (id) DO UPDATE SET
                properties = $4
        """, edge_id, source_id, target_id, json.dumps(edge_data))
```

```python
class Neo4JStorage(BaseGraphStorage):
    def __init__(self, namespace, global_config, **kwargs):
        super().__init__(namespace, global_config)
        self.driver = None

    async def initialize(self):
        self.driver = GraphDatabase.driver(
            os.getenv("NEO4J_URI"),
            auth=(os.getenv("NEO4J_USERNAME"), os.getenv("NEO4J_PASSWORD"))
        )

    async def upsert_node(self, node_id, node_data):
        async with self.driver.session() as session:
            await session.run(
                "MERGE (n:Entity {id: $id}) SET n += $properties",
                id=node_id, properties=node_data
            )

    async def upsert_edge(self, source_id, target_id, edge_data):
        async with self.driver.session() as session:
            await session.run(
                """
                MATCH (a:Entity {id: $source}), (b:Entity {id: $target})
                MERGE (a)-[r:RELATES]->(b)
                SET r += $properties
                """,
                source=source_id, target=target_id, properties=edge_data
            )

    async def get_node(self, node_id):
        async with self.driver.session() as session:
            result = await session.run(
                "MATCH (n:Entity {id: $id}) RETURN n",
                id=node_id
            )
            record = await result.single()
            return dict(record["n"]) if record else None

    async def get_node_edges(self, node_id):
        async with self.driver.session() as session:
            result = await session.run(
                """
                MATCH (n:Entity {id: $id})-[r]-(m)
                RETURN m.id as target, type(r) as rel_type
                """,
                id=node_id
            )
            records = await result.data()
            return [(r["target"], r["rel_type"]) for r in records]
```

```python
    async def initialize(self):

    async def get_by_id(self, id):
        doc = await self.collection.find_one({"key": id})
        return doc["value"] if doc else None

    async def upsert(self, data):
        operations = [
            UpdateOne(
                {"key": key},
                {"$set": {"key": key, "value": value}},
                upsert=True
            )
            for key, value in data.items()
        ]
        await self.collection.bulk_write(operations)
```

```python
class RedisKVStorage(BaseKVStorage):
    def __init__(self, namespace, global_config, **kwargs):
        super().__init__(namespace, global_config)
        self.prefix = f"lightrag:{namespace}:"

    async def initialize(self):
        self.redis = await aioredis.from_url(
            os.getenv("REDIS_URI"),
            encoding="utf-8",
            decode_responses=True
        )

    async def get_by_id(self, id):
        data = await self.redis.get(f"{self.prefix}{id}")
        return json.loads(data) if data else None

    async def upsert(self, data):
        pipe = self.redis.pipeline()
        for key, value in data.items():
            pipe.set(f"{self.prefix}{key}", json.dumps(value))
        await pipe.execute()
```

```python
class MilvusVectorDBStorage(BaseVectorStorage):
    def __init__(self, namespace, global_config, embedding_func, **kwargs):
        super().__init__(namespace, global_config, embedding_func)
        self.collection_name = f"lightrag_vector_{namespace}"

    async def initialize(self):
        connections.connect(
            alias="default",
            host=os.getenv("MILVUS_URI"),
            port=os.getenv("MILVUS_PORT", "19530")
        )

        # 创建集合
        fields = [
            FieldSchema(name="id", dtype=DataType.VARCHAR, max_length=256, is_primary=True),
            FieldSchema(name="vector", dtype=DataType.FLOAT_VECTOR, dim=self.embedding_func.embedding_dim),
            FieldSchema(name="metadata", dtype=DataType.JSON),
        ]
        schema = CollectionSchema(fields=fields)
        self.collection = Collection(self.collection_name, schema)

        # 创建索引
        index_params = {
            "metric_type": "COSINE",
            "index_type": "IVF_FLAT",
            "params": {"nlist": 1024}
        }
        self.collection.create_index("vector", index_params)

    async def query(self, query, top_k=10, ids=None):
        if isinstance(query, str):
            query = await self.embedding_func([query])
            query = query[0]

        self.collection.load()
        search_params = {"metric_type": "COSINE", "params": {"nprobe": 16}}

        results = self.collection.search(
            data=[query],
            anns_field="vector",
            param=search_params,
            limit=top_k,
            output_fields=["metadata"]
        )

        return [
            {"id": hit.entity.get("id"), "distance": hit.distance, "metadata": hit.entity.get("metadata")}
            for hit in results[0]
        ]
```

```python
class QdrantVectorDBStorage(BaseVectorStorage):

            limit=top_k,
            query_filter=Filter(must=[FieldCondition(key="id", match=MatchAny(any=ids))]) if ids else None

            {"id": hit.id, "distance": hit.score, "metadata": hit.payload}
```

```python
class FaissVectorDBStorage(BaseVectorStorage):
    def __init__(self, namespace, global_config, embedding_func, **kwargs):
        super().__init__(namespace, global_config, embedding_func)
        self.index = None
        self.id_map = {}
        self.file_path = os.path.join(global_config["working_dir"], f"{namespace}.faiss")

    async def initialize(self):
        if os.path.exists(self.file_path):
            self.index = faiss.read_index(self.file_path)
        else:
            self.index = faiss.IndexFlatIP(self.embedding_func.embedding_dim)

    async def query(self, query, top_k=10, ids=None):
        if isinstance(query, str):
            query = await self.embedding_func([query])
            query = query[0]

        query = np.array([query]).astype("float32")
        faiss.normalize_L2(query)

        distances, indices = self.index.search(query, top_k)

        results = []
        for i, (dist, idx) in enumerate(zip(distances[0], indices[0])):
            if idx != -1 and idx in self.id_map:
                results.append({
                    "id": self.id_map[idx],
                    "distance": float(dist)
                })
        return results

    async def upsert(self, data):
        ids = list(data.keys())
        vectors = np.array([d["vector"] for d in data.values()]).astype("float32")
        faiss.normalize_L2(vectors)

        start_id = self.index.ntotal
        self.index.add(vectors)

        for i, id in enumerate(ids):
            self.id_map[start_id + i] = id

        faiss.write_index(self.index, self.file_path)
```

| 无外部依赖 | 不支持分布式 |
| 多种索引类型 | 无过滤能力 |
| 内存高效 | 仅支持向量 |

```python
class OpenSearchKVStorage(BaseKVStorage):
    def __init__(self, namespace, global_config, **kwargs):
        super().__init__(namespace, global_config)
        self.index_name = f"lightrag_kv_{namespace}"

    async def initialize(self):
        self.client = OpenSearch(
            hosts=json.loads(os.getenv("OPENSEARCH_HOSTS")),
            http_auth=(
                os.getenv("OPENSEARCH_USER"),
                os.getenv("OPENSEARCH_PASSWORD")
            ),
        )
        # 创建索引
        if not self.client.indices.exists(index=self.index_name):
            self.client.indices.create(index=self.index_name)

    async def get_by_id(self, id):
        try:
            result = self.client.get(index=self.index_name, id=id)
            return result["_source"]
        except NotFoundError:
            return None

    async def upsert(self, data):
        actions = []
        for key, value in data.items():
            actions.append({"index": {"_index": self.index_name, "_id": key}})
            actions.append(value)
        self.client.bulk(body=actions)
```

```python
class MemgraphStorage(BaseGraphStorage):
    def __init__(self, namespace, global_config, **kwargs):
        super().__init__(namespace, global_config)

    async def initialize(self):
        self.connection = Memgraph(host=os.getenv("MEMGRAPH_URI", "localhost"), port=7687)

    async def upsert_node(self, node_id, node_data):
        self.connection.execute(
            "MERGE (n:Entity {id: $id}) SET n += $properties",
            {"id": node_id, "properties": node_data}
        )

    async def upsert_edge(self, source_id, target_id, edge_data):
        self.connection.execute(
            """
            MATCH (a:Entity {id: $source}), (b:Entity {id: $target})
            MERGE (a)-[r:RELATES]->(b)
            SET r += $properties
            """,
            {"source": source_id, "target": target_id, "properties": edge_data}
        )
```
