### 1.3 工作流步骤 1: start_exploration (workflow.py)

```python
@step
async def start_exploration(self, ev: InputEvent, ctx: Context[WorkflowState], agent: FsExplorerAgent):
    # 1. 保存状态
    async with ctx.store.edit_state() as state:
        state.intial_task = ev.task  # "找到 data/ 目录下的 testfile.txt 并总结"

    # 2. 描述当前目录
    dirdescription = describe_dir_content(".")
    # 输出: "Content of .\nFILES:\n- ./src\n- ./tests\n- ./data\n..."

    # 3. 配置 Agent 任务
    agent.configure_task(
        f"Given that the current directory ('.') looks like this:\n\n```text\n{dirdescription}```\n\n"
        f"And that the user is giving you this task: '{ev.task}', what action should you take first?"
    )

    # 4. 获取 Agent 决策
    result = await agent.take_action()
```

### 1.4 Agent 决策 (agent.py)

```python
async def take_action(self) -> tuple[Action, ActionType] | None:
    # 1. 发送对话历史到 Gemini
    response = await self._client.aio.models.generate_content(
        model="gemini-3-flash-preview",
        contents=self._chat_history,
        config={
            "response_mime_type": "application/json",
            "response_json_schema": Action.model_json_schema(),
        },
    )

    # 2. 解析响应
    action = Action.model_validate_json(response.text)
    # 假设返回: {"action": {"tool_name": "glob", "tool_input": [{"parameter_name": "directory", "parameter_value": "."}, {"parameter_name": "pattern", "parameter_value": "**/testfile.txt"}]}, "reason": "Searching for testfile.txt"}

    # 3. 执行工具
    if action.to_action_type() == "toolcall":
        await self.call_tool(tool_name="glob", tool_input={"directory": ".", "pattern": "**/testfile.txt"})
```

### 1.5 工具执行 (agent.py → fs.py)

```python
async def call_tool(self, tool_name: Tools, tool_input: dict[str, Any]) -> None:
    result = TOOLS["glob"](**{"directory": ".", "pattern": "**/testfile.txt"})
    # glob_paths(".", "**/testfile.txt") → "MATCHES for **/testfile.txt in .:\n\n- ./data/testfile.txt"

    # 追加到对话历史
    self._chat_history.append(
        Content(role="user", parts=[Part.from_text(text="Tool result for glob:\n\nMATCHES for **/testfile.txt in .:\n\n- ./data/testfile.txt")])
    )
```

### 1.6 下一步决策

Agent 基于工具结果继续决策，可能返回 `toolcall read`：

```python

action = Action(
```

### 1.7 文件读取 (fs.py)

```python
def read_file(file_path: str) -> str:

        return f.read()
```
